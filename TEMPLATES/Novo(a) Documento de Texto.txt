<script>
    async function enviar(e) {
        if (e.key === "Enter") {
            let input = document.getElementById("input");
            let mensagem = input.value.trim();
            if (!mensagem) return;

            mostrar("Você", mensagem, "user");
            input.value = "";

            // Chamada para o Flask
            let resposta = await fetch("/chat", {
                method: "POST",
                headers: { "Content-Type": "application/json" },
                body: JSON.stringify({ mensagem })
            });

            let data = await resposta.json();
            mostrar("APRIME", data.resposta, "bot");
        }
    }

    function mostrar(nome, texto, classe) {
        let messages = document.getElementById("messages");
        let div = document.createElement("div");
        div.className = "msg " + classe;

        // Destaque para sugestões
        if(texto.startsWith("Aqui vai uma sugestão:")) {
            div.innerHTML = `<b>${nome}:</b> <i>${texto}</i>`;
        } else {
            div.innerHTML = `<b>${nome}:</b> ${texto}`;
        }

        messages.appendChild(div);
        messages.scrollTop = messages.scrollHeight;
    }
</script>
